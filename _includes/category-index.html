{% comment %} PlebWare automatic master-category index. Discovers one-level subcategory index.md pages and shows the three newest posts from each subcategory. Posts are matched inside the loop so the current subcategory path is evaluated directly. This avoids relying on a dynamically scoped variable inside the where_exp expression and prevents cross-category matches. The shared include is used by all mode indexes. Usage: {% include category-index.html section_slug="graphics" %} {% endcomment %} {% assign section_slug = include.section_slug %} {% assign section_prefix = section_slug | append: '/' %} {% assign subcategory_pages = site.pages | sort: 'path' %} {% for subpage in subcategory_pages %} {% assign path_parts = subpage.path | split: '/' %} {% if subpage.path contains section_prefix and path_parts.size == 3 and path_parts[0] == section_slug and path_parts[2] == 'index.md' %} {% assign subcategory_path = subpage.path | remove: '/index.md' | append: '/' %} {% assign subcategory_posts = '' | split: '' %} {% for post in site.posts %} {% if post.path contains subcategory_path %} {% assign subcategory_posts = subcategory_posts | push: post %} {% endif %} {% endfor %} {% assign subcategory_posts = subcategory_posts | sort: 'date' | reverse %} ## {{ subpage.title | default: path_parts[1] | escape }} {% if subcategory_posts.size > 0 %}
    {% for post in subcategory_posts limit: 3 %}
  • {{ post.title | escape }} – {{ post.date | date: "%Y-%m-%d" }}
    • {% endfor %}
{% else %}

No articles yet.

{% endif %} {% endif %} {% endfor %}
